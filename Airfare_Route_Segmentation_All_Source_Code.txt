    print(f"Exported to {route_csv_path}")

    # 3. Prepare Scaled Clustering Matrix
    X_scaled, scaler, feature_cols = prepare_clustering_matrix(route_df)
    print(f"Clustering Matrix Shape: {X_scaled.shape} across {len(feature_cols)} behavioural features:")
    for f in feature_cols:
        print(f"  - {f}")

    # 4. Multi-K Hyperparameter Sweep (WCSS, Silhouette, Davies-Bouldin)
    print("\n[Step 4/7] Evaluating K-Means Across Multiple Cluster Counts (K=2 to 8)...")
    k_eval_df = evaluate_kmeans_sweep(X_scaled, k_range=[2, 3, 4, 5, 6, 7, 8])
    print(k_eval_df.to_string(index=False))

    # 5. Distance Metric Comparison (Euclidean vs Manhattan)
    print("\n[Step 5/7] Comparing Euclidean vs Manhattan Distance Formulations (K=3, 4, 5)...")
    dist_comp_df = compare_distance_metrics(X_scaled, k_values=[3, 4, 5])
    print(dist_comp_df.to_string(index=False))

    # 6. Cluster Stability & Selection of Defensible K
    chosen_k = 4
    print(f"\n[Step 6/7] Defensible Clustering Selected: K={chosen_k}")
    stability_res = evaluate_cluster_stability(X_scaled, k=chosen_k, n_runs=10)
    print(f"Stability (10 Random Seed Initializations):")
    print(f"  Mean Adjusted Rand Index (ARI): {stability_res['mean_ari']} (+/- {stability_res['std_ari']})")
    print(f"  Mean Normalized Mutual Info (NMI): {stability_res['mean_nmi']} (+/- {stability_res['std_nmi']})")

    # Final Fit on chosen K
    final_kmeans = run_kmeans_clustering(X_scaled, k=chosen_k, random_state=42)
    cluster_labels = final_kmeans.labels_
    route_df["cluster"] = cluster_labels

    # PCA Dimensionality Reduction
    pca, X_pca, exp_var = compute_pca_projection(X_scaled, n_components=3)
    route_df["pca_1"] = X_pca[:, 0]
    route_df["pca_2"] = X_pca[:, 1]
    route_df["pca_3"] = X_pca[:, 2]

    print(f"PCA Variance Explained: PC1={exp_var[0]*100:.1f}%, PC2={exp_var[1]*100:.1f}%, PC3={exp_var[2]*100:.1f}% (Total: {np.sum(exp_var)*100:.1f}%)")

    # Cluster Archetypes & Profiling
    archetypes = generate_cluster_archetypes(route_df, cluster_labels, k=chosen_k)
    route_df["archetype_name"] = route_df["cluster"].map(lambda c: archetypes[c]["archetype_name"])

    # Export labeled assignments
    assignments_path = os.path.join(DATA_DIR, "route_cluster_assignments.csv")
    route_df.to_csv(assignments_path, index=False)
    print(f"Saved clustered route assignments to {assignments_path}")

    # 7. Export Comprehensive Model Metadata for Web Application & Reports
    print("\n[Step 7/7] Generating Cluster Interpretations, Limitations & Ethical Document...")
    silhouette_dist = compute_silhouette_distribution(X_scaled, cluster_labels)
    ethical_doc = generate_ethical_and_limitations_report()

    # PCA feature loadings
    loadings = pd.DataFrame(
        pca.components_.T,
        columns=["PC1", "PC2", "PC3"],
        index=feature_cols
    ).round(4).to_dict(orient="index")

    # PCA points serialization
    pca_points = []
    for idx, row in route_df.iterrows():
        pca_points.append({
            "route": row["route"],
            "cluster": int(row["cluster"]),
            "archetype": row["archetype_name"],
            "pc1": round(float(row["pca_1"]), 3),
            "pc2": round(float(row["pca_2"]), 3),
            "pc3": round(float(row["pca_3"]), 3),
            "mean_price": float(row["mean_price"]),
            "avg_duration": float(row["avg_duration"]),
            "surge_ratio": float(row["last_minute_surge_ratio"])
        })

    model_metadata = {
        "chosen_k": chosen_k,
        "k_eval_sweep": k_eval_df.to_dict(orient="records"),
        "distance_comparison": dist_comp_df.to_dict(orient="records"),
        "stability": stability_res,
        "silhouette_distribution": silhouette_dist,
        "pca_variance_explained": [round(float(v), 4) for v in exp_var],
        "pca_loadings": loadings,
        "pca_points": pca_points,
        "archetypes": archetypes,
        "ethical_considerations": ethical_doc["ethical_considerations"],
        "limitations": ethical_doc["limitations"],
        "feature_names": feature_cols
    }

    metrics_json_path = os.path.join(DATA_DIR, "model_metrics.json")
    with open(metrics_json_path, "w") as f:
        json.dump(model_metadata, f, indent=4)
    print(f"Saved complete metrics payload to {metrics_json_path}")

    print("\n" + "=" * 80)
    print("PIPELINE EXECUTION COMPLETED SUCCESSFULLY!")
    print("=" * 80)

if __name__ == "__main__":
    run()

def get_route_distance(src, dst):
    if src == dst:
        return 400
    if (src, dst) in ROUTE_DISTANCES_KM:
        return ROUTE_DISTANCES_KM[(src, dst)]
    if (dst, src) in ROUTE_DISTANCES_KM:
        return ROUTE_DISTANCES_KM[(dst, src)]
    return 1450

def generate_flight_data(num_samples: int = 70000, random_seed: int = 42) -> pd.DataFrame:
    """
    Generates rich, comprehensive flight observations across 70+ corridors
    incorporating realistic yield management, class differentials, and carrier distributions.
    """
    np.random.seed(random_seed)
    records = []

    # Assemble comprehensive route list
    route_pool = []
    for k in ROUTE_DISTANCES_KM.keys():
        src, dst = k
        is_intl = src in INTERNATIONAL_CITIES or dst in INTERNATIONAL_CITIES
        route_pool.append((src, dst, is_intl))
        route_pool.append((dst, src, is_intl))

    for idx in range(num_samples):
        src, dst, is_intl = route_pool[np.random.randint(0, len(route_pool))]

        if is_intl:
            airline = np.random.choice(
                ["Air_India", "Vistara", "British_Airways", "Emirates", "Singapore_Airlines", "Air_France", "Lufthansa", "Indigo"],
                p=[0.20, 0.18, 0.14, 0.15, 0.12, 0.08, 0.08, 0.05]
            )
        else:
            airline = np.random.choice(
                ["Indigo", "Air_India", "Vistara", "SpiceJet", "AirAsia", "GO_FIRST"],
                p=[0.38, 0.22, 0.18, 0.10, 0.08, 0.04]
            )

        prefix = {
            "Indigo": "6E", "Air_India": "AI", "Vistara": "UK", "SpiceJet": "SG",
            "AirAsia": "I5", "GO_FIRST": "G8", "British_Airways": "BA", "Emirates": "EK",
            "Singapore_Airlines": "SQ", "Air_France": "AF", "Lufthansa": "LH"
        }.get(airline, "FL")
        flight_code = f"{prefix}-{np.random.randint(100, 9999)}"

        dep_time = np.random.choice(TIME_SLOTS, p=[0.18, 0.25, 0.18, 0.22, 0.13, 0.04])
        arr_time = np.random.choice(TIME_SLOTS, p=[0.15, 0.23, 0.20, 0.23, 0.15, 0.04])

        dist_km = get_route_distance(src, dst)

        # Stop probability
        if dist_km > 5000:
            stops = np.random.choice(STOPS_OPTIONS, p=[0.35, 0.52, 0.13])
        elif dist_km < 600:
            stops = np.random.choice(STOPS_OPTIONS, p=[0.82, 0.17, 0.01])
        else:
            stops = np.random.choice(STOPS_OPTIONS, p=[0.42, 0.52, 0.06])

        # Duration
        base_flight_hours = dist_km / 780.0
        if stops == "zero":
            duration = round(base_flight_hours + np.random.uniform(0.3, 0.7), 2)
        elif stops == "one":
            layover = np.random.uniform(1.5, 5.5)
            duration = round(base_flight_hours * 1.15 + layover, 2)
        else:
            layover = np.random.uniform(4.0, 10.0)
            duration = round(base_flight_hours * 1.3 + layover, 2)

        # Cabin Class
        if airline in ["SpiceJet", "AirAsia", "GO_FIRST"]:
            travel_class = "Economy"
        elif airline == "Indigo":
            travel_class = np.random.choice(["Economy", "Business"], p=[0.90, 0.10])
        else:
            travel_class = np.random.choice(["Economy", "Business"], p=[0.65, 0.35])

        # Days Left (1 to 50)
        days_left = int(np.random.randint(1, 51))

        # Yield Management Pricing
        base_rate = 3.5 if not is_intl else 4.6
        raw_base_fare = max(1800, dist_km * base_rate * AIRLINE_BASE_COST.get(airline, 1.0))

        class_mult = 1.0 if travel_class == "Economy" else np.random.uniform(3.5, 4.9)

        # Non-linear surge curve
        if days_left <= 3:
            surge_mult = np.random.uniform(1.85, 2.65)
        elif days_left <= 7:
            surge_mult = np.random.uniform(1.45, 1.95)
        elif days_left <= 15:
            surge_mult = np.random.uniform(1.15, 1.45)
        elif days_left <= 32:
            surge_mult = np.random.uniform(0.92, 1.12)
        else:
            surge_mult = np.random.uniform(0.85, 1.02)

        time_mult = 1.12 if dep_time in ["Morning", "Evening"] else (0.88 if dep_time in ["Late_Night", "Night"] else 1.0)
        stops_mult = 1.08 if stops == "zero" else (0.95 if stops == "one" else 0.88)

        noise = np.random.normal(1.0, 0.06)

        records.append({
            "Unnamed: 0": idx,
            "airline": airline,
            "flight": flight_code,
            "source_city": src,
            "departure_time": dep_time,
            "stops": stops,
            "arrival_time": arr_time,
            "destination_city": dst,
            "class": travel_class,
            "duration": duration,
            "days_left": days_left,
            "price": final_price
        })

    return pd.DataFrame(records)

    return df

"""
Feature Engineering and Preprocessing Module for Flight Price Prediction & Route Segmentation.
Performs:
1. Handling missing values (median/mode imputation).
2. Categorical variable encoding (Ordinal & One-Hot).
3. Route-level historical pricing behavior feature aggregation.
4. Numerical scaling (StandardScaler & RobustScaler).
5. Generation of modeling matrices for K-Means clustering.
"""

import os
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler, RobustScaler

DATA_DIR = os.path.join(os.path.dirname(os.path.dirname(__file__)), "data")

def preprocess_flight_records(df: pd.DataFrame) -> pd.DataFrame:
    """
    Cleans raw flight records, handles missing values, and creates standard derived variables.
    """
    df = df.copy()

    # Drop index column if present
    if "Unnamed: 0" in df.columns:
        df = df.drop(columns=["Unnamed: 0"])

    # 1. Missing Value Check & Imputation
    num_cols = df.select_dtypes(include=[np.number]).columns
    cat_cols = df.select_dtypes(include=["object"]).columns

    for col in num_cols:
        if df[col].isnull().sum() > 0:
            df[col] = df[col].fillna(df[col].median())

    for col in cat_cols:
        if df[col].isnull().sum() > 0:
            df[col] = df[col].fillna(df[col].mode()[0])

    # 2. Derive Route identifier
    df["route"] = df["source_city"] + " - " + df["destination_city"]

    # 3. Ordinal mapping for stops & class
    stops_map = {"zero": 0, "one": 1, "two_or_more": 2}
    df["stops_num"] = df["stops"].map(stops_map).fillna(1)

    class_map = {"Economy": 0, "Business": 1}
    df["class_num"] = df["class"].map(class_map).fillna(0)

    # 4. Booking advance windows
    df["is_last_minute"] = (df["days_left"] <= 3).astype(int)
    df["is_early_bird"] = (df["days_left"] >= 21).astype(int)
    df["is_peak_hour"] = df["departure_time"].isin(["Morning", "Evening"]).astype(int)

    return df

def aggregate_route_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Aggregates historical flight pricing behavior at the route corridor level.
    Constructs behavioral dimensions:
    - Mean & Median Price
    - Price Standard Deviation & Coefficient of Variation (Volatility)
    - Last-Minute Surge Ratio (Yield Management Aggressiveness)
    - Early Bird Savings Gradient
    - Price Spread (90th - 10th Percentile)
    - Business Class Share & Spread
    - Direct Flight Ratio & Average Duration
    - Airline Competition Index
    """
    routes = []

    for route_name, group in df.groupby("route"):
        src = group["source_city"].iloc[0]
        dst = group["destination_city"].iloc[0]
        sample_count = len(group)

        # Baseline pricing metrics
        mean_price = group["price"].mean()
        median_price = group["price"].median()
        price_std = group["price"].std() if sample_count > 1 else 0
        price_cv = (price_std / mean_price) if mean_price > 0 else 0

        # Percentiles
        p10 = group["price"].quantile(0.10)
        p25 = group["price"].quantile(0.25)
        p75 = group["price"].quantile(0.75)
        p90 = group["price"].quantile(0.90)
        price_iqr = p75 - p25
        price_spread = p90 - p10

        # Last minute pricing behavior (days_left <= 3 vs days_left >= 21)
        last_minute = group[group["days_left"] <= 3]["price"]
        advance_book = group[group["days_left"] >= 21]["price"]

        last_min_mean = last_minute.mean() if len(last_minute) > 0 else mean_price * 1.5
        advance_mean = advance_book.mean() if len(advance_book) > 0 else mean_price * 0.9
        last_minute_surge_ratio = last_min_mean / advance_mean if advance_mean > 0 else 1.5

        # Flight duration & Direct flight ratio
        avg_duration = group["duration"].mean()
        direct_ratio = (group["stops_num"] == 0).mean()

        # Business class share & premium multiplier
        business_ratio = (group["class_num"] == 1).mean()
        econ_mean = group[group["class_num"] == 0]["price"].mean() if (group["class_num"] == 0).any() else mean_price
        biz_mean = group[group["class_num"] == 1]["price"].mean() if (group["class_num"] == 1).any() else mean_price * 3.5
        business_premium_ratio = (biz_mean / econ_mean) if econ_mean > 0 else 3.5

        # Peak hour departure premium
        peak_mean = group[group["is_peak_hour"] == 1]["price"].mean() if (group["is_peak_hour"] == 1).any() else mean_price
        offpeak_mean = group[group["is_peak_hour"] == 0]["price"].mean() if (group["is_peak_hour"] == 0).any() else mean_price

"""
Clustering and Dimensionality Reduction Module.
Implements:
1. Standard K-Means (Euclidean Distance).
2. Manhattan-Distance K-Medians / L1 Clustering.
3. Multi-K hyperparameter sweep (K=2 through K=8, focusing on K=3, 4, 5).
4. Principal Component Analysis (PCA) 2D/3D projections and biplot loading analysis.
5. Cluster centroid profiling and assignment.
"""

import numpy as np
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.metrics import pairwise_distances

class ManhattanKMedians:
    """
    K-Medians clustering using Manhattan (L1) distance metric.
    Robust to extreme airfare outliers and asymmetric price spikes.
    """
    def __init__(self, n_clusters: int = 4, max_iter: int = 300, tol: float = 1e-4, random_state: int = 42):
        self.n_clusters = n_clusters
        self.max_iter = max_iter
        self.tol = tol
        self.random_state = random_state
        self.cluster_centers_ = None
        self.labels_ = None
        self.inertia_ = None

    def fit(self, X: np.ndarray):
        np.random.seed(self.random_state)
        n_samples, n_features = X.shape

        # Initialize medians randomly from existing data points
        init_indices = np.random.choice(n_samples, self.n_clusters, replace=False)
        self.cluster_centers_ = X[init_indices].copy()

        for iteration in range(self.max_iter):
            # Compute Manhattan (L1) distance from all samples to cluster centers
            # shape: (n_samples, n_clusters)
            dist_matrix = pairwise_distances(X, self.cluster_centers_, metric="manhattan")
            labels = np.argmin(dist_matrix, axis=1)

            new_centers = np.zeros_like(self.cluster_centers_)
            for k in range(self.n_clusters):
                cluster_points = X[labels == k]
                if len(cluster_points) > 0:
                    # Component-wise median minimizes L1 distance
                    new_centers[k] = np.median(cluster_points, axis=0)
                else:
                    # If empty cluster, re-initialize to a random data point
                    new_centers[k] = X[np.random.choice(n_samples)]

            center_shift = np.sum(np.abs(self.cluster_centers_ - new_centers))
            self.cluster_centers_ = new_centers

            if center_shift < self.tol:
                break

        self.labels_ = labels
        # Calculate L1 inertia (sum of Manhattan distances to nearest center)
        final_dists = pairwise_distances(X, self.cluster_centers_, metric="manhattan")
        self.inertia_ = np.sum(np.min(final_dists, axis=1))
        return self

    def predict(self, X: np.ndarray) -> np.ndarray:
        dist_matrix = pairwise_distances(X, self.cluster_centers_, metric="manhattan")
        return np.argmin(dist_matrix, axis=1)

def run_kmeans_clustering(X: np.ndarray, k: int, random_state: int = 42) -> KMeans:
    """
    Fits standard K-Means clustering with Euclidean distance.
    """
    kmeans = KMeans(
        n_clusters=k,
        init="k-means++",
        n_init=15,
        max_iter=300,
        random_state=random_state
    )
    kmeans.fit(X)
    return kmeans

def run_manhattan_clustering(X: np.ndarray, k: int, random_state: int = 42) -> ManhattanKMedians:
    """
    Fits K-Medians clustering using Manhattan (L1) distance.
    """
    kmedians = ManhattanKMedians(
        n_clusters=k,
        max_iter=300,
        tol=1e-4,
        random_state=random_state
    )
    kmedians.fit(X)
    return kmedians

def compute_pca_projection(X: np.ndarray, n_components: int = 3):
    """
    Applies PCA dimensionality reduction for visual inspection.
    Returns:
    - pca model
    - X_pca (projected coordinates)
    - explained_variance_ratio
    - loadings DataFrame
    """
    pca = PCA(n_components=n_components, random_state=42)
    X_pca = pca.fit_transform(X)
    explained_variance = pca.explained_variance_ratio_

    return pca, X_pca, explained_variance

"""
Model Evaluation and Validation Module for Unsupervised Clustering.
Computes:
1. Within-Cluster Sum of Squares (WCSS / Inertia / Elbow Method).
2. Silhouette Coefficient (Overall mean & per-cluster samples).
3. Davies-Bouldin Index (Cluster separation to scatter ratio).
4. Distance Metric Comparison (Euclidean vs Manhattan).
5. Cluster Stability Analysis across random initializations (ARI & NMI).
"""

import numpy as np
import pandas as pd
from sklearn.metrics import (
    silhouette_score,
    silhouette_samples,
    davies_bouldin_score,
    adjusted_rand_score,
    normalized_mutual_info_score,
    pairwise_distances
)
from clustering import run_kmeans_clustering, run_manhattan_clustering

def evaluate_kmeans_sweep(X: np.ndarray, k_range: list = [2, 3, 4, 5, 6, 7, 8], random_state: int = 42) -> pd.DataFrame:
    """
    Evaluates K-Means across a spectrum of cluster counts K.
    Calculates WCSS (Inertia), Silhouette Score, and Davies-Bouldin Index for each K.
    """
    results = []

    for k in k_range:
        kmeans = run_kmeans_clustering(X, k=k, random_state=random_state)
        labels = kmeans.labels_

        wcss = kmeans.inertia_
        sil = silhouette_score(X, labels, metric="euclidean")
        db = davies_bouldin_score(X, labels)

        results.append({
            "k": k,
            "wcss": round(wcss, 2),
            "silhouette": round(sil, 4),
            "davies_bouldin": round(db, 4)
        })

    return pd.DataFrame(results)

def compare_distance_metrics(X: np.ndarray, k_values: list = [3, 4, 5], random_state: int = 42) -> pd.DataFrame:
    """
    Compares Euclidean Distance (K-Means) against Manhattan Distance (K-Medians).
    Evaluates:
    - Silhouette Score (computed under Euclidean and Manhattan metrics)
    - Davies-Bouldin Index
    - Cluster size balance / dispersion
    """
    comparison = []

    for k in k_values:
        # 1. Euclidean K-Means
        kmeans = run_kmeans_clustering(X, k=k, random_state=random_state)
        euc_labels = kmeans.labels_
        euc_sil_euc = silhouette_score(X, euc_labels, metric="euclidean")
        euc_sil_man = silhouette_score(X, euc_labels, metric="manhattan")
        euc_db = davies_bouldin_score(X, euc_labels)

        comparison.append({
            "k": k,
            "algorithm": "K-Means (Euclidean L2)",
            "silhouette_euclidean": round(euc_sil_euc, 4),
            "silhouette_manhattan": round(euc_sil_man, 4),
            "davies_bouldin": round(euc_db, 4),
            "inertia": round(kmeans.inertia_, 2)
        })

        # 2. Manhattan K-Medians
        kmedians = run_manhattan_clustering(X, k=k, random_state=random_state)
        man_labels = kmedians.labels_
        man_sil_euc = silhouette_score(X, man_labels, metric="euclidean")
        man_sil_man = silhouette_score(X, man_labels, metric="manhattan")
        man_db = davies_bouldin_score(X, man_labels)

        comparison.append({
            "k": k,
            "algorithm": "K-Medians (Manhattan L1)",
            "silhouette_euclidean": round(man_sil_euc, 4),
            "silhouette_manhattan": round(man_sil_man, 4),
            "davies_bouldin": round(man_db, 4),
            "inertia": round(kmedians.inertia_, 2)
        })

    return pd.DataFrame(comparison)

def evaluate_cluster_stability(X: np.ndarray, k: int = 4, n_runs: int = 10) -> dict:
    """
    Tests clustering solution stability across varying random initializations.
    Computes pairwise Adjusted Rand Index (ARI) and Normalized Mutual Information (NMI).
    ARI close to 1.0 indicates strong deterministic partition stability.
    """
    all_labels = []
    for seed in range(n_runs):
        km = run_kmeans_clustering(X, k=k, random_state=seed * 17 + 3)
        all_labels.append(km.labels_)

    ari_scores = []
    nmi_scores = []

    for i in range(n_runs):
        for j in range(i + 1, n_runs):
            ari_scores.append(adjusted_rand_score(all_labels[i], all_labels[j]))
            nmi_scores.append(normalized_mutual_info_score(all_labels[i], all_labels[j]))

    return {
        "k": k,
        "n_runs": n_runs,
        "mean_ari": round(float(np.mean(ari_scores)), 4),
        "std_ari": round(float(np.std(ari_scores)), 4),
        "mean_nmi": round(float(np.mean(nmi_scores)), 4),
        "std_nmi": round(float(np.std(nmi_scores)), 4)
    }

def compute_silhouette_distribution(X: np.ndarray, labels: np.ndarray) -> dict:
    """
    Computes per-cluster silhouette scores to detect weak or overlapping clusters.
    """
    samples = silhouette_samples(X, labels)
    k = len(np.unique(labels))
    cluster_scores = {}

    for c in range(k):
        sub_samples = samples[labels == c]
        cluster_scores[c] = {
            "mean_silhouette": round(float(np.mean(sub_samples)), 4),
            "min_silhouette": round(float(np.min(sub_samples)), 4),
            "max_silhouette": round(float(np.max(sub_samples)), 4),
            "sample_count": int(len(sub_samples))
        }

    return cluster_scores

    def resolve_city_name(self, city_or_code: str) -> str:
        clean = (city_or_code or "").strip().upper()
        if clean in CITY_CODE_MAP:
            return CITY_CODE_MAP[clean]
        for name in CITY_CODE_MAP.values():
            if name.upper() == clean:
                return name
        return city_or_code.title()

    def get_route_record(self, origin: str, destination: str):
        origin_city = self.resolve_city_name(origin)
        dest_city = self.resolve_city_name(destination)
        route_name = f"{origin_city} - {dest_city}"

        if not self.routes_df.empty:
            match = self.routes_df[self.routes_df["route"] == route_name]
            if not match.empty:
                return match.iloc[0].to_dict()

            rev_route = f"{dest_city} - {origin_city}"
            rev_match = self.routes_df[self.routes_df["route"] == rev_route]
            if not rev_match.empty:
                rec = rev_match.iloc[0].to_dict()
                rec["route"] = route_name
                rec["source_city"] = origin_city
                rec["destination_city"] = dest_city
                return rec

        # Dynamic fallback
        is_intl = origin_city in ["London", "Dubai", "Singapore", "New York", "Paris", "Bangkok", "Tokyo", "Frankfurt"] or \
                  dest_city in ["London", "Dubai", "Singapore", "New York", "Paris", "Bangkok", "Tokyo", "Frankfurt"]

    def predict_booking_decision(self, origin: str, destination: str, depart_date_str: str = None, travel_class: str = "Economy"):
        origin_city = self.resolve_city_name(origin)
        dest_city = self.resolve_city_name(destination)
        route_rec = self.get_route_record(origin_city, dest_city)

        cluster_id = str(int(route_rec.get("cluster", 0)))
        archetype_info = self.metrics.get("archetypes", {}).get(cluster_id, {})
        archetype_name = archetype_info.get("archetype_name", route_rec.get("archetype_name", "Commercial Corridor"))

        days_left = 25
        if depart_date_str:
            try:
                now_date = datetime.date(2026, 11, 15)
                parts = depart_date_str.split("-")
                dep_date = datetime.date(int(parts[0]), int(parts[1]), int(parts[2]))
                diff = (dep_date - now_date).days
                days_left = max(1, diff)
            except Exception:
                days_left = 25

        class_factor = 3.8 if travel_class.lower() == "business" else 1.0

        base_median = route_rec.get("median_price", 6000.0) * class_factor
        base_p10 = route_rec.get("p10_price", 4000.0) * class_factor
        base_p25 = route_rec.get("p25_price", 5000.0) * class_factor
        base_p75 = route_rec.get("p75_price", 11000.0) * class_factor
        base_p90 = route_rec.get("p90_price", 18000.0) * class_factor
        base_mean = route_rec.get("mean_price", 8500.0) * class_factor

        surge_ratio = route_rec.get("last_minute_surge_ratio", 2.0)

        # Recommendation logic
        if days_left <= 3:
            current_fare = base_median * (surge_ratio * 1.08)
            verdict = "BOOK NOW"
            verdict_badge = "book-now"
            verdict_summary = "Emergency Surge Alert: Flight is in departure-week critical surge window. Seats are evaporating quickly."
            confidence = 97
        elif days_left <= 7:
            current_fare = base_median * (surge_ratio * 0.85)
            verdict = "BOOK NOW"
            verdict_badge = "book-now"
            verdict_summary = f"Inside steep price climb zone ({days_left} days left). Booking now avoids the severe last-minute surge of +{int((surge_ratio-1)*100)}%."
            confidence = 94
        elif 16 <= days_left <= 34:
            current_fare = base_p25 * 1.02
            verdict = "BOOK NOW"
            verdict_badge = "book-now"
            verdict_summary = f"Optimal 'Goldilocks Window' Active ({days_left} days left). Fares are in the lowest historical 25th percentile. Lock in now."
            confidence = 93
        elif days_left > 45:
            current_fare = base_p75 * 0.96
            verdict = "WAIT"
            verdict_badge = "wait"
            verdict_summary = f"Booking {days_left} days ahead is premature. Airline yield systems release discounted fare buckets ~21–35 days before departure."
            confidence = 89
        else:
            current_fare = base_median
            verdict = "TYPICAL / MONITOR"
            verdict_badge = "monitor"
            verdict_summary = f"Fares are stable near median ({days_left} days left). Set an alert and pounce if fares dip below INR {int(base_p25):,}."
            confidence = 86

        price_range = max(1, base_p90 - base_p10)
        percentile_rank = int(np.clip(((current_fare - base_p10) / price_range) * 100, 5, 95))

        # Expected surge peak
        peak_surge_price = int(base_median * surge_ratio * 1.05)
        potential_savings = max(0, peak_surge_price - int(current_fare))
        savings_pct = int(round((potential_savings / peak_surge_price) * 100)) if peak_surge_price > 0 else 0

        # Booking curve points
        curve_days = [50, 45, 40, 35, 30, 25, 21, 14, 10, 7, 5, 3, 2, 1]
        curve_prices = []
        for d in curve_days:
            if d > 35:
                mult = 1.05
            elif d >= 18:
                mult = 0.92
            elif d >= 8:
                mult = 1.15
            elif d >= 4:
                mult = surge_ratio * 0.75
            else:
                mult = surge_ratio * 1.05
            curve_prices.append(round(base_median * mult, 0))

        origin_meta = CITY_META.get(origin_city, {"code": origin_city[:3].upper(), "airport": f"{origin_city} Airport", "category": "City"})
        dest_meta = CITY_META.get(dest_city, {"code": dest_city[:3].upper(), "airport": f"{dest_city} Airport", "category": "City"})

        # Generate realistic flight options
        flight_options = self.generate_flight_options(
            origin_city=origin_city,
            dest_city=dest_city,
            current_fare=current_fare,
            travel_class=travel_class,
            avg_duration=route_rec.get("avg_duration", 3.5)
        )

        return {
            "origin": origin_city,
            "origin_code": origin_meta["code"],
            "origin_airport": origin_meta["airport"],
            "destination": dest_city,
            "destination_code": dest_meta["code"],
            "destination_airport": dest_meta["airport"],
            "route": route_rec.get("route"),

            "verdict_badge": verdict_badge,
            "verdict_summary": verdict_summary,

            "estimated_fare": int(current_fare),
            "peak_surge_fare": peak_surge_price,
            "potential_savings_inr": potential_savings,

            "percentile_rank": percentile_rank,

/**
 * AEROMATRIX Cockpit Controller
 */

document.addEventListener("DOMContentLoaded", () => {
  // Mode / View Switcher
  const viewBtns = document.querySelectorAll(".nav-tab-btn");
  const viewPanels = document.querySelectorAll(".view-panel");

  viewBtns.forEach(btn => {
    btn.addEventListener("click", () => {
      const targetView = btn.getAttribute("data-view");
      viewBtns.forEach(b => b.classList.remove("active"));
      viewPanels.forEach(p => p.classList.remove("active"));

      btn.classList.add("active");
      const activePanel = document.getElementById(targetView);
      if (activePanel) {
        activePanel.classList.add("active");
      }
    });
  });

  // Search Form
  const searchForm = document.getElementById("cockpitSearchForm");
  if (searchForm) {
    searchForm.addEventListener("submit", (e) => {
      e.preventDefault();
      executeCockpitScan();
    });
  }

  // Airport Swap Button
  const btnSwap = document.getElementById("btnSwapAtc");
  if (btnSwap) {
    btnSwap.addEventListener("click", () => {
      const fromSel = document.getElementById("fromCity");
      const toSel = document.getElementById("toCity");
      if (fromSel && toSel) {
        const temp = fromSel.value;
        fromSel.value = toSel.value;
        toSel.value = temp;

        btnSwap.style.transform = "rotate(180deg)";
        setTimeout(() => { btnSwap.style.transform = "none"; }, 300);

        executeCockpitScan();
      }
    });
  }

  // Quick Vector Presets
  document.querySelectorAll(".preset-pill").forEach(pill => {
    pill.addEventListener("click", () => {
      const from = pill.getAttribute("data-from");
      const to = pill.getAttribute("data-to");
      const fromSel = document.getElementById("fromCity");
      const toSel = document.getElementById("toCity");

      if (fromSel && toSel) {
        fromSel.value = from;
        toSel.value = to;
        executeCockpitScan();
      }
    });
  });

  // PCA Route Filter
  const pcaFilter = document.getElementById("pcaRouteFilter");
  if (pcaFilter) {
    pcaFilter.addEventListener("input", (e) => {
      const query = (e.target.value || "").trim().toLowerCase();
      if (window.MODEL_METRICS && window.MODEL_METRICS.pca_points) {
        let pts = window.MODEL_METRICS.pca_points;
        if (query) {
          pts = pts.filter(p => p.route.toLowerCase().includes(query));
        }
        renderPcaScatter(pts, window.CURRENT_ACTIVE_ROUTE || "");
      }
    });
  }

  // Initialize with initial data
  if (window.INITIAL_DATA) {
    window.CURRENT_ACTIVE_ROUTE = window.INITIAL_DATA.route;
    updateCockpitUI(window.INITIAL_DATA);
  }

  if (window.MODEL_METRICS) {
    renderEvaluationHub(window.MODEL_METRICS);
  }
});

function executeCockpitScan() {
  const fromCity = document.getElementById("fromCity").value;
  const toCity = document.getElementById("toCity").value;
  const departDate = document.getElementById("departDate").value;
  const travelClass = document.getElementById("travelClass").value;

  const btn = document.querySelector(".btn-execute-analysis");
  const origHtml = btn.innerHTML;
  btn.innerHTML = `<span>⏳ Scanning Corridors...</span>`;
  btn.disabled = true;

  const url = `/api/predict?from=${encodeURIComponent(fromCity)}&to=${encodeURIComponent(toCity)}&depart=${encodeURIComponent(departDate)}&class=${encodeURIComponent(travelClass)}`;

  fetch(url)
    .then(res => res.json())
    .then(data => {
      window.CURRENT_ACTIVE_ROUTE = data.route;
      updateCockpitUI(data);
    })
    .catch(err => {
      console.error("Telemetry query failed:", err);
    })
    .finally(() => {
      btn.innerHTML = origHtml;
      btn.disabled = false;
    });
}

function updateCockpitUI(data) {
  // 1. Route IATA Callout
  const calloutIata = document.getElementById("calloutIata");
  if (calloutIata) {
    calloutIata.innerHTML = `<span>${data.origin_code}</span> <span class="flight-arrow">➔</span> <span>${data.destination_code}</span>`;
  }

import os
import sys
import unittest
import numpy as np

from clustering import (
    run_kmeans_clustering,
    run_manhattan_clustering,
    compute_pca_projection
)
from evaluation import evaluate_kmeans_sweep, compare_distance_metrics

class TestClustering(unittest.TestCase):
    def setUp(self):
        np.random.seed(42)
        # 3 artificial clusters in 5-dimensional space
        c1 = np.random.normal(loc=-3, scale=0.5, size=(15, 5))
        c2 = np.random.normal(loc=0, scale=0.5, size=(15, 5))
        c3 = np.random.normal(loc=4, scale=0.5, size=(15, 5))
        self.X = np.vstack([c1, c2, c3])

    def test_kmeans_euclidean(self):
        km = run_kmeans_clustering(self.X, k=3, random_state=42)
        self.assertEqual(len(np.unique(km.labels_)), 3)
        self.assertGreater(km.inertia_, 0)

    def test_manhattan_clustering(self):
        kmed = run_manhattan_clustering(self.X, k=3, random_state=42)
        self.assertEqual(len(np.unique(kmed.labels_)), 3)
        self.assertGreater(kmed.inertia_, 0)

    def test_pca_projection(self):
        pca, X_pca, exp_var = compute_pca_projection(self.X, n_components=3)
        self.assertEqual(X_pca.shape, (45, 3))
        self.assertEqual(len(exp_var), 3)
        self.assertAlmostEqual(float(np.sum(exp_var)), 1.0, delta=0.2)

    def test_evaluation_sweep(self):
        res_df = evaluate_kmeans_sweep(self.X, k_range=[2, 3, 4])
        self.assertEqual(len(res_df), 3)
        self.assertIn("silhouette", res_df.columns)
        self.assertIn("davies_bouldin", res_df.columns)

if __name__ == "__main__":
    unittest.main()
