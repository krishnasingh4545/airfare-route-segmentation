def get_route_distance(src, dst):
    if src == dst:
        return 400
    if (src, dst) in ROUTE_DISTANCES_KM:
        return ROUTE_DISTANCES_KM[(src, dst)]

        return ROUTE_DISTANCES_KM[(dst, src)]

"""
Feature Engineering and Preprocessing Module for Flight Price Prediction & Route Segmentation.
Performs:
1. Handling missing values (median/mode imputation).
2. Categorical variable encoding (Ordinal & One-Hot).
3. Route-level historical pricing behavior feature aggregation.
4. Numerical scaling (StandardScaler & RobustScaler).
5. Generation of modeling matrices for K-Means clustering.
"""

import os
import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler, RobustScaler

def preprocess_flight_records(df: pd.DataFrame) -> pd.DataFrame:
    """
    Cleans raw flight records, handles missing values, and creates standard derived variables.
    """
    df = df.copy()

    # Drop index column if present
    if "Unnamed: 0" in df.columns:
        df = df.drop(columns=["Unnamed: 0"])

    # 1. Missing Value Check & Imputation
    num_cols = df.select_dtypes(include=[np.number]).columns
    cat_cols = df.select_dtypes(include=["object"]).columns

    for col in num_cols:
        if df[col].isnull().sum() > 0:
            df[col] = df[col].fillna(df[col].median())

    for col in cat_cols:
        if df[col].isnull().sum() > 0:
            df[col] = df[col].fillna(df[col].mode()[0])

    # 2. Derive Route identifier
    df["route"] = df["source_city"] + " - " + df["destination_city"]

    # 3. Ordinal mapping for stops & class
    stops_map = {"zero": 0, "one": 1, "two_or_more": 2}
    df["stops_num"] = df["stops"].map(stops_map).fillna(1)

    class_map = {"Economy": 0, "Business": 1}
    df["class_num"] = df["class"].map(class_map).fillna(0)

    # 4. Booking advance windows
    df["is_last_minute"] = (df["days_left"] <= 3).astype(int)
    df["is_early_bird"] = (df["days_left"] >= 21).astype(int)
    df["is_peak_hour"] = df["departure_time"].isin(["Morning", "Evening"]).astype(int)

    return df

def aggregate_route_features(df: pd.DataFrame) -> pd.DataFrame:
    """
    Aggregates historical flight pricing behavior at the route corridor level.
    Constructs behavioral dimensions:
    - Mean & Median Price
    - Price Standard Deviation & Coefficient of Variation (Volatility)
    - Last-Minute Surge Ratio (Yield Management Aggressiveness)
    - Early Bird Savings Gradient
    - Price Spread (90th - 10th Percentile)
    - Business Class Share & Spread
    - Direct Flight Ratio & Average Duration
    - Airline Competition Index
    """
    routes = []

    for route_name, group in df.groupby("route"):
        src = group["source_city"].iloc[0]
        dst = group["destination_city"].iloc[0]
        sample_count = len(group)

        # Baseline pricing metrics
        mean_price = group["price"].mean()
        median_price = group["price"].median()
        price_std = group["price"].std() if sample_count > 1 else 0
        price_cv = (price_std / mean_price) if mean_price > 0 else 0

        # Percentiles
        p10 = group["price"].quantile(0.10)
        p25 = group["price"].quantile(0.25)
        p75 = group["price"].quantile(0.75)
        p90 = group["price"].quantile(0.90)
        price_iqr = p75 - p25
        price_spread = p90 - p10

        # Last minute pricing behavior (days_left <= 3 vs days_left >= 21)
        last_minute = group[group["days_left"] <= 3]["price"]
        advance_book = group[group["days_left"] >= 21]["price"]

        last_min_mean = last_minute.mean() if len(last_minute) > 0 else mean_price * 1.5
        advance_mean = advance_book.mean() if len(advance_book) > 0 else mean_price * 0.9
        last_minute_surge_ratio = last_min_mean / advance_mean if advance_mean > 0 else 1.5

        # Flight duration & Direct flight ratio
        avg_duration = group["duration"].mean()
        direct_ratio = (group["stops_num"] == 0).mean()

        # Business class share & premium multiplier
        business_ratio = (group["class_num"] == 1).mean()
        econ_mean = group[group["class_num"] == 0]["price"].mean() if (group["class_num"] == 0).any() else mean_price
        biz_mean = group[group["class_num"] == 1]["price"].mean() if (group["class_num"] == 1).any() else mean_price * 3.5
        business_premium_ratio = (biz_mean / econ_mean) if econ_mean > 0 else 3.5

        # Peak hour departure premium
        peak_mean = group[group["is_peak_hour"] == 1]["price"].mean() if (group["is_peak_hour"] == 1).any() else mean_price
        offpeak_mean = group[group["is_peak_hour"] == 0]["price"].mean() if (group["is_peak_hour"] == 0).any() else mean_price
        peak_premium_ratio = (peak_mean / offpeak_mean) if offpeak_mean > 0 else 1.1

        # Carrier competition
        unique_airlines = group["airline"].nunique()

        routes.append({
            "route": route_name,
            "source_city": src,
            "destination_city": dst,
            "sample_count": sample_count,
            "mean_price": round(mean_price, 2),

"""
Clustering and Dimensionality Reduction Module.
Implements:

3. Multi-K hyperparameter sweep (K=2 through K=8, focusing on K=3, 4, 5).

5. Cluster centroid profiling and assignment.
"""

import numpy as np
import pandas as pd
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.metrics import pairwise_distances

class ManhattanKMedians:
    """
    K-Medians clustering using Manhattan (L1) distance metric.
    Robust to extreme airfare outliers and asymmetric price spikes.
    """
    def __init__(self, n_clusters: int = 4, max_iter: int = 300, tol: float = 1e-4, random_state: int = 42):
        self.n_clusters = n_clusters
        self.max_iter = max_iter
        self.tol = tol
        self.random_state = random_state
        self.cluster_centers_ = None
        self.labels_ = None
        self.inertia_ = None

    def fit(self, X: np.ndarray):
        np.random.seed(self.random_state)
        n_samples, n_features = X.shape

        # Initialize medians randomly from existing data points
        init_indices = np.random.choice(n_samples, self.n_clusters, replace=False)
        self.cluster_centers_ = X[init_indices].copy()

        for iteration in range(self.max_iter):
            # Compute Manhattan (L1) distance from all samples to cluster centers
            # shape: (n_samples, n_clusters)
            dist_matrix = pairwise_distances(X, self.cluster_centers_, metric="manhattan")
            labels = np.argmin(dist_matrix, axis=1)

            new_centers = np.zeros_like(self.cluster_centers_)
            for k in range(self.n_clusters):
                cluster_points = X[labels == k]
                if len(cluster_points) > 0:
                    # Component-wise median minimizes L1 distance
                    new_centers[k] = np.median(cluster_points, axis=0)
                else:
                    # If empty cluster, re-initialize to a random data point
                    new_centers[k] = X[np.random.choice(n_samples)]

            center_shift = np.sum(np.abs(self.cluster_centers_ - new_centers))
            self.cluster_centers_ = new_centers

            if center_shift < self.tol:
                break

        self.labels_ = labels
        # Calculate L1 inertia (sum of Manhattan distances to nearest center)
        final_dists = pairwise_distances(X, self.cluster_centers_, metric="manhattan")
        self.inertia_ = np.sum(np.min(final_dists, axis=1))
        return self

    def predict(self, X: np.ndarray) -> np.ndarray:
        dist_matrix = pairwise_distances(X, self.cluster_centers_, metric="manhattan")
        return np.argmin(dist_matrix, axis=1)

def evaluate_kmeans_sweep(X: np.ndarray, k_range: list = [2, 3, 4, 5, 6, 7, 8], random_state: int = 42) -> pd.DataFrame:

def evaluate_cluster_stability(X: np.ndarray, k: int = 4, n_runs: int = 10) -> dict:
    """
    Tests clustering solution stability across varying random initializations.
    Computes pairwise Adjusted Rand Index (ARI) and Normalized Mutual Information (NMI).
    ARI close to 1.0 indicates strong deterministic partition stability.
    """
    all_labels = []
    for seed in range(n_runs):
        km = run_kmeans_clustering(X, k=k, random_state=seed * 17 + 3)
        all_labels.append(km.labels_)

    ari_scores = []
    nmi_scores = []

    for i in range(n_runs):
        for j in range(i + 1, n_runs):
            ari_scores.append(adjusted_rand_score(all_labels[i], all_labels[j]))
            nmi_scores.append(normalized_mutual_info_score(all_labels[i], all_labels[j]))

    return {
        "k": k,
        "n_runs": n_runs,
        "mean_ari": round(float(np.mean(ari_scores)), 4),
        "std_ari": round(float(np.std(ari_scores)), 4),
        "mean_nmi": round(float(np.mean(nmi_scores)), 4),
        "std_nmi": round(float(np.std(nmi_scores)), 4)
    }

document.addEventListener("DOMContentLoaded", () => {

  const viewBtns = document.querySelectorAll(".nav-tab-btn");
  const viewPanels = document.querySelectorAll(".view-panel");

  viewBtns.forEach(btn => {
    btn.addEventListener("click", () => {
      const targetView = btn.getAttribute("data-view");
      viewBtns.forEach(b => b.classList.remove("active"));
      viewPanels.forEach(p => p.classList.remove("active"));

      btn.classList.add("active");
      const activePanel = document.getElementById(targetView);
      if (activePanel) {
        activePanel.classList.add("active");
      }
    });
  });

  // Search Form
  const searchForm = document.getElementById("cockpitSearchForm");
  if (searchForm) {
    searchForm.addEventListener("submit", (e) => {
      e.preventDefault();
      executeCockpitScan();
    });
  }

  // Airport Swap Button
  const btnSwap = document.getElementById("btnSwapAtc");
  if (btnSwap) {
    btnSwap.addEventListener("click", () => {
      const fromSel = document.getElementById("fromCity");
      const toSel = document.getElementById("toCity");
      if (fromSel && toSel) {
        const temp = fromSel.value;
        fromSel.value = toSel.value;
        toSel.value = temp;

        btnSwap.style.transform = "rotate(180deg)";
        setTimeout(() => { btnSwap.style.transform = "none"; }, 300);

        executeCockpitScan();
      }
    });
  }

  // Quick Vector Presets
  document.querySelectorAll(".preset-pill").forEach(pill => {
    pill.addEventListener("click", () => {
      const from = pill.getAttribute("data-from");
      const to = pill.getAttribute("data-to");
      const fromSel = document.getElementById("fromCity");
      const toSel = document.getElementById("toCity");

      if (fromSel && toSel) {
        fromSel.value = from;
        toSel.value = to;
        executeCockpitScan();
      }
    });
  });

  // PCA Route Filter
  const pcaFilter = document.getElementById("pcaRouteFilter");
  if (pcaFilter) {
    pcaFilter.addEventListener("input", (e) => {
      const query = (e.target.value || "").trim().toLowerCase();
      if (window.MODEL_METRICS && window.MODEL_METRICS.pca_points) {
        let pts = window.MODEL_METRICS.pca_points;
        if (query) {
          pts = pts.filter(p => p.route.toLowerCase().includes(query));
        }
        renderPcaScatter(pts, window.CURRENT_ACTIVE_ROUTE || "");
      }
    });
  }

  // Initialize with initial data
  if (window.INITIAL_DATA) {
    window.CURRENT_ACTIVE_ROUTE = window.INITIAL_DATA.route;
    updateCockpitUI(window.INITIAL_DATA);
  }

  if (window.MODEL_METRICS) {
    renderEvaluationHub(window.MODEL_METRICS);
  }
});

function executeCockpitScan() {
  const fromCity = document.getElementById("fromCity").value;
  const toCity = document.getElementById("toCity").value;
  const departDate = document.getElementById("departDate").value;
  const travelClass = document.getElementById("travelClass").value;

  const btn = document.querySelector(".btn-execute-analysis");
  const origHtml = btn.innerHTML;
  btn.innerHTML = `<span>⏳ Scanning Corridors...</span>`;
  btn.disabled = true;

  const url = `/api/predict?from=${encodeURIComponent(fromCity)}&to=${encodeURIComponent(toCity)}&depart=${encodeURIComponent(departDate)}&class=${encodeURIComponent(travelClass)}`;

  fetch(url)
    .then(res => res.json())
    .then(data => {
      window.CURRENT_ACTIVE_ROUTE = data.route;
      updateCockpitUI(data);
    })
    .catch(err => {
      console.error("Telemetry query failed:", err);
    })
    .finally(() => {
      btn.innerHTML = origHtml;
      btn.disabled = false;
    });
}

function updateCockpitUI(data) {

  }

  const verdictBadge = document.getElementById("verdictBadge");
  if (verdictBadge) {
    verdictBadge.className = `signal-badge-cockpit ${data.verdict_badge}`;
    const icon = data.verdict_badge === "book-now" ? "⚡" : (data.verdict_badge === "wait" ? "⏳" : "👁");
    verdictBadge.innerHTML = `<span>${icon}</span> ${data.verdict}`;
  }

  // 3. Telemetry Cells
  const statFare = document.getElementById("statFare");
  if (statFare) statFare.textContent = `₹${data.estimated_fare.toLocaleString()}`;

  const statConfidence = document.getElementById("statConfidence");
